   400→   400→export function TasksView() {

   402→   402→  const {

   409→   409→  } = useTaskStore();
   410→   410→
   411→   411→  // filter / search / sort state
   412→   412→  const [search, setSearch] = React.useState("");
   413→   413→  const [categoryFilter, setCategoryFilter] = React.useState<
   414→   414→    "all" | TaskCategory
   415→   415→  >("all");
   416→   416→  const [priorityFilter, setPriorityFilter] = React.useState<"all" | Priority>(
   417→   417→    "all"
   418→   418→  );
   419→   419→  const [sort, setSort] = React.useState<SortKey>("due-asc");
   420→   420→
   421→   421→  // dialog state
   422→   422→  const [dialogOpen, setDialogOpen] = React.useState(false);
   423→   423→  const [editingTask, setEditingTask] = React.useState<Task | undefined>(
   424→   424→    undefined
   425→   425→  );
   426→   426→
   427→   427→  // ---- stats ----
   428→   428→  const total = tasks.length;
   429→   429→  const completedCount = tasks.filter((t) => t.completed).length;
   430→   430→  const pendingCount = total - completedCount;
   431→   431→  const overdueCount = tasks.filter(isOverdue).length;
   432→   432→  const dueTodayCount = tasks.filter(
   433→   433→    (t) => !t.completed && isDueToday(t)
   434→   434→  ).length;
   435→   435→
   436→   436→  // ---- derived filtered + sorted list ----
   437→   437→  const filtered = React.useMemo(() => {
   438→   438→    const q = search.trim().toLowerCase();
   439→   439→    const list = tasks.filter((t) => {
   440→   440→      if (categoryFilter !== "all" && t.category !== categoryFilter)
   441→   441→        return false;
   442→   442→      if (priorityFilter !== "all" && t.priority !== priorityFilter)
   443→   443→        return false;
   444→   444→      if (q) {
   445→   445→        const hay = `${t.title}\n${t.description}`.toLowerCase();
   446→   446→        if (!hay.includes(q)) return false;
   447→   447→      }
   448→   448→      return true;
   449→   449→    });
   450→   450→
   451→   451→    return [...list].sort((a, b) => {
   452→   452→      switch (sort) {
   453→   453→        case "due-asc":
   454→   454→          // earlier dates first; overdue (past) dates naturally sort before today
   455→   455→          return a.dueDate.localeCompare(b.dueDate);
   456→   456→        case "due-desc":
   457→   457→          return b.dueDate.localeCompare(a.dueDate);
   458→   458→        case "priority-desc":
   459→   459→          return PRIORITY_RANK[b.priority] - PRIORITY_RANK[a.priority];
   460→   460→        case "recent":
   461→   461→          return b.createdAt.localeCompare(a.createdAt);
   462→   462→        default:
   463→   463→          return 0;
   464→   464→      }
   465→   465→    });
   466→   466→  }, [tasks, search, categoryFilter, priorityFilter, sort]);
   467→   467→
   468→   468→  const hasActiveFilters =
   469→   469→    search.trim() !== "" ||
   470→   470→    categoryFilter !== "all" ||
   471→   471→    priorityFilter !== "all";
   472→   472→
   473→   473→  function resetFilters() {
   474→   474→    setSearch("");
   475→   475→    setCategoryFilter("all");
   476→   476→    setPriorityFilter("all");
   477→   477→    setSort("due-asc");
   478→   478→  }
   479→   479→
   480→   480→  function openAdd() {

   483→   483→  }

   485→   485→    setEditingTask(task);
   486→   486→    setDialogOpen(true);
   487→   487→  }
   488→   488→
   489→   489→  function handleSubmit(state: FormState) {
   490→   490→    const input = formStateToInput(state);
   491→   491→    if (editingTask) {
   492→   492→      updateTask(editingTask.id, input);
   493→   493→      toast.success("Task updated");
   494→   494→    } else {
   495→   495→      addTask(input);
   496→   496→      toast.success("Task added");
   497→   497→    }
   498→   498→    setDialogOpen(false);
   499→   499→    setEditingTask(undefined);
   500→   500→  }
   501→   501→
   502→   502→  function handleToggle(task: Task) {
   503→   503→    const next = !task.completed;
   504→   504→    toggleTask(task.id);
   505→   505→    toast.success(next ? "Task completed" : "Marked pending");
   506→   506→  }
   507→   507→
   508→   508→  function handleDelete(id: string) {
   509→   509→    deleteTask(id);
   510→   510→    toast.success("Task deleted");
   511→   511→  }
   512→   512→
   513→   513→  function handleClearCompleted() {
   514→   514→    const n = completedCount;
   515→   515→    if (n === 0) return;
   516→   516→    clearCompleted();
   517→   517→    toast.success(
   518→   518→      `${n} completed task${n === 1 ? "" : "s"} cleared`
   519→   519→    );
   520→   520→  }
   521→   521→
   522→   522→  // SSR-safe placeholder until the persisted store hydrates.
   523→   523→  if (!hydrated) return <TasksSkeleton />;
   524→   524→
   525→   525→  return (
   526→   526→    <div className="space-y-6">
   527→   527→      {/* Header */}
   528→   528→      <ModuleHeader
   529→   529→        icon={<ListTodo className="h-5 w-5" />}
   530→   530→        title="Tasks"
   531→   531→        description="Your assignments and to-dos"
   532→   532→        actions={
   533→   533→          <>
   534→   534→            {completedCount > 0 ? (
   535→   535→              <Button
