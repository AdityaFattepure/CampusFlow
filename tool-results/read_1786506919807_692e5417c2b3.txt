    90→    90→type TimeFilter = "month" | "week" | "all";
    91→    91→type SortKey = "newest" | "oldest" | "amount-desc" | "amount-asc";
    92→    92→type CategoryFilter = "all" | ExpenseCategory;

    94→    94→/** Whether an ISO yyyy-mm-dd falls within the last 7 calendar days (inclusive of today). */
    95→    95→function withinThisWeek(iso: string, ref: Date = new Date()): boolean {
    96→    96→  if (!iso) return false;
    97→    97→  const target = new Date(iso + "T00:00:00");
    98→    98→  const today = new Date(ref.getFullYear(), ref.getMonth(), ref.getDate());
    99→    99→  const diffDays = Math.round(
   100→   100→    (today.getTime() - target.getTime()) / (1000 * 60 * 60 * 24)
   101→   101→  );
   102→   102→  return diffDays >= 0 && diffDays <= 6;
   103→   103→}
   104→   104→
   105→   105→function matchesTimeFilter(iso: string, filter: TimeFilter): boolean {
   106→   106→  if (filter === "all") return true;
   107→   107→  if (filter === "month") return isSameMonth(iso);
   108→   108→  return withinThisWeek(iso);
   109→   109→}

   111→   111→function sortExpenses(list: Expense[], key: SortKey): Expense[] {
   112→   112→  const copy = [...list];
   113→   113→  switch (key) {
   114→   114→    case "oldest":
   115→   115→      copy.sort(
   116→   116→        (a, b) =>
   117→   117→          a.date.localeCompare(b.date) || a.createdAt.localeCompare(b.createdAt)
   118→   118→      );
   119→   119→      break;
   120→   120→    case "amount-desc":
   121→   121→      copy.sort((a, b) => b.amount - a.amount);
   122→   122→      break;
   123→   123→    case "amount-asc":
   124→   124→      copy.sort((a, b) => a.amount - b.amount);
   125→   125→      break;
   126→   126→    case "newest":
   127→   127→    default:
   128→   128→      copy.sort(
   129→   129→        (a, b) =>
   130→   130→          b.date.localeCompare(a.date) || b.createdAt.localeCompare(a.createdAt)
   131→   131→      );
   132→   132→      break;
   133→   133→  }
   134→   134→  return copy;
   135→   135→}
