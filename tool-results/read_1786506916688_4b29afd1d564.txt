    90→type TimeFilter = "month" | "week" | "all";
    91→type SortKey = "newest" | "oldest" | "amount-desc" | "amount-asc";
    92→type CategoryFilter = "all" | ExpenseCategory;

    94→/** Whether an ISO yyyy-mm-dd falls within the last 7 calendar days (inclusive of today). */
    95→function withinThisWeek(iso: string, ref: Date = new Date()): boolean {
    96→  if (!iso) return false;
    97→  const target = new Date(iso + "T00:00:00");
    98→  const today = new Date(ref.getFullYear(), ref.getMonth(), ref.getDate());
    99→  const diffDays = Math.round(
   100→    (today.getTime() - target.getTime()) / (1000 * 60 * 60 * 24)
   101→  );
   102→  return diffDays >= 0 && diffDays <= 6;
   103→}
   104→
   105→function matchesTimeFilter(iso: string, filter: TimeFilter): boolean {
   106→  if (filter === "all") return true;
   107→  if (filter === "month") return isSameMonth(iso);
   108→  return withinThisWeek(iso);
   109→}
   110→
   111→function sortExpenses(list: Expense[], key: SortKey): Expense[] {
   112→  const copy = [...list];
   113→  switch (key) {
   114→    case "oldest":
   115→      copy.sort(
   116→        (a, b) =>
   117→          a.date.localeCompare(b.date) || a.createdAt.localeCompare(b.createdAt)
   118→      );
   119→      break;
   120→    case "amount-desc":
   121→      copy.sort((a, b) => b.amount - a.amount);
   122→      break;
   123→    case "amount-asc":
   124→      copy.sort((a, b) => a.amount - b.amount);
   125→      break;
   126→    case "newest":
   127→    default:
   128→      copy.sort(
   129→        (a, b) =>
   130→          b.date.localeCompare(a.date) || b.createdAt.localeCompare(a.createdAt)
   131→      );
   132→      break;
   133→  }
   134→  return copy;
   135→}

   162→function ExpenseFormDialog({
   163→  trigger,
   164→  initial,
   165→  onSubmit,
   166→  submitLabel,
   167→  title,
   168→  description,
   169→}: {
   170→  trigger: React.ReactNode;
   171→  initial?: Expense;
   172→  onSubmit: (input: ExpenseInput) => void;
   173→  submitLabel: string;
   174→  title: string;
   175→  description: string;
   176→}) {
   177→  const [open, setOpen] = React.useState(false);
   178→  const [form, setForm] = React.useState<FormState>(() =>
   179→    buildFormState(initial)
   180→  );
   181→
   182→  // Reset the form whenever the dialog is reopened so it reflects the latest
   183→  // `initial` value (important for the edit use case).
   184→  React.useEffect(() => {
   185→    if (open) setForm(buildFormState(initial));
   186→  }, [open, initial]);
   187→
   188→  const amountNum = Number(form.amount);
   189→  const amountValid =
   190→    form.amount.trim() !== "" && Number.isFinite(amountNum) && amountNum > 0;
   191→
   192→  function handleSubmit(e: React.FormEvent) {
   193→    e.preventDefault();
   194→    if (!amountValid) {
   195→      toast.error("Enter a valid amount");
   196→      return;
   197→    }
   198→    onSubmit({
   199→      amount: Math.round(amountNum),
   200→      category: form.category,
   201→      description: form.description.trim(),
   202→      date: form.date || todayISO(),
   203→    });
   204→    setOpen(false);
   205→  }
   206→
   207→  return (
   208→    <Dialog open={open} onOpenChange={setOpen}>
   209→      <DialogTrigger asChild>{trigger}</DialogTrigger>
   210→      <DialogContent className="sm:max-w-md">
   211→        <DialogHeader>
   212→          <DialogTitle>{title}</DialogTitle>
   213→          <DialogDescription>{description}</DialogDescription>
   214→        </DialogHeader>
   215→        <form onSubmit={handleSubmit} className="space-y-4">
   216→          <div className="space-y-2">
   217→            <Label htmlFor="exp-amount">Amount</Label>
   218→            <div className="relative">
   219→              <IndianRupee className="pointer-events-none absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   220→              <Input
   221→                id="exp-amount"
   222→                type="number"
   223→                min={1}
   224→                step="1"
   225→                inputMode="numeric"
   226→                placeholder="0"
   227→                className="pl-8"
   228→                value={form.amount}
   229→                onChange={(e) =>
   230→                  setForm((f) => ({ ...f, amount: e.target.value }))
   231→                }
   232→                autoFocus
   233→                aria-invalid={!amountValid && form.amount !== ""}
   234→              />
   235→            </div>
   236→            {!amountValid && form.amount !== "" ? (
   237→              <p className="text-xs text-destructive">
   238→                Enter an amount greater than zero.
   239→              </p>
   240→            ) : null}
   241→          </div>
   242→
   243→          <div className="grid grid-cols-2 gap-3">
   244→            <div className="space-y-2">
   245→              <Label htmlFor="exp-category">Category</Label>
   246→              <Select
   247→                value={form.category}
   248→                onValueChange={(v) =>
   249→                  setForm((f) => ({ ...f, category: v as ExpenseCategory }))
   250→                }
   251→              >
   252→                <SelectTrigger id="exp-category" className="w-full">
   253→                  <SelectValue />
   254→                </SelectTrigger>
   255→                <SelectContent>
   256→                  {EXPENSE_CATEGORIES.map((c) => (
   257→                    <SelectItem key={c} value={c}>
   258→                      {c}
   259→                    </SelectItem>
   260→                  ))}
   261→                </SelectContent>
   262→              </Select>
   263→            </div>
   264→            <div className="space-y-2">
   265→              <Label htmlFor="exp-date">Date</Label>
   266→              <Input
   267→                id="exp-date"
   268→                type="date"
   269→                value={form.date}
   270→                max={todayISO()}
   271→                onChange={(e) =>
   272→                  setForm((f) => ({ ...f, date: e.target.value }))
   273→                }
   274→              />
   275→            </div>
   276→          </div>
   277→
   278→          <div className="space-y-2">
   279→            <Label htmlFor="exp-desc">Description</Label>
   280→            <Textarea
   281→              id="exp-desc"
   282→              placeholder="Optional note about this expense..."
   283→              rows={2}
   284→              value={form.description}
   285→              onChange={(e) =>
   286→                setForm((f) => ({ ...f, description: e.target.value }))
   287→              }
   288→            />
   289→          </div>
   290→
   291→          <DialogFooter className="pt-2">
   292→            <DialogClose asChild>
   293→              <Button type="button" variant="outline">
   294→                Cancel
   295→              </Button>
   296→            </DialogClose>
   297→            <Button type="submit" disabled={!amountValid}>
   298→              {submitLabel}
   299→            </Button>
   300→          </DialogFooter>
   301→        </form>
   302→      </DialogContent>
   303→    </Dialog>
   304→  );
   305→}

   339→  const [sortKey, setSortKey] = React.useState<SortKey>("newest");
   340→
   341→  if (!hydrated) return <ExpensesSkeleton />;
   342→
   343→  // --- filter + sort the visible set ---
   344→  const q = search.trim().toLowerCase();
   345→  const filtered = expenses.filter((e) => {
   346→    if (!matchesTimeFilter(e.date, timeFilter)) return false;
   347→    if (categoryFilter !== "all" && e.category !== categoryFilter) return false;
   348→    if (q && !e.description.toLowerCase().includes(q)) return false;
   349→    return true;
   350→  });
   351→  const sorted = sortExpenses(filtered, sortKey);
   352→
   353→  // --- totals ---
   354→  const filteredTotal = filtered.reduce((s, e) => s + e.amount, 0);
   355→  const monthExpenses = expenses.filter((e) => isSameMonth(e.date));
   356→  const monthTotal = monthExpenses.reduce((s, e) => s + e.amount, 0);
   357→  const allTimeTotal = expenses.reduce((s, e) => s + e.amount, 0);
   358→
   359→  // --- per-category breakdown for current filter set ---
   360→  const byCategory = new Map<ExpenseCategory, number>();
   361→  for (const e of filtered) {
   362→    byCategory.set(e.category, (byCategory.get(e.category) ?? 0) + e.amount);
   363→  }
   364→  const breakdown = [...byCategory.entries()]
   365→    .map(([cat, value]) => ({ cat, value }))
   366→    .sort((a, b) => b.value - a.value);
   367→
   368→  // --- top category this month (for stat card) ---
   369→  const monthByCategory = new Map<ExpenseCategory, number>();
   370→  for (const e of monthExpenses) {
   371→    monthByCategory.set(
   372→      e.category,
   373→      (monthByCategory.get(e.category) ?? 0) + e.amount
   374→    );
   375→  }
   376→  const topMonthEntry = [...monthByCategory.entries()].sort(
   377→    (a, b) => b[1] - a[1]
   378→  )[0];
   379→
   380→  // --- chart data + config (only categories present in filtered set) ---
   381→  const chartData = breakdown.map(({ cat, value }) => ({
   382→    name: cat,
   383→    value,
   384→    fill: EXPENSE_CATEGORY_META[cat].color,
   385→  }));

   387→    ...Object.fromEntries(
   388→      breakdown.map(({ cat }) => [

   390→        {
   391→          label: cat,

   406→      title="Add Expense"
   407→      description="Log a new spending entry."
   408→      submitLabel="Add Expense"
   409→      onSubmit={(input) => {
   410→        addExpense(input);
   411→        toast.success("Expense added");
   412→      }}
   413→    />
   414→  );
   415→
   416→  return (
   417→    <div className="space-y-6">
   418→      <ModuleHeader
   419→        icon={<Wallet className="h-5 w-5" />}
   420→        title="Expenses"
   421→        description="Track where your money goes"

   427→        <StatCard
   428→          icon={<Wallet className="h-4 w-4" />}
   429→          label={`This Month — ${monthLabel()}`}
   430→          value={formatINR(monthTotal)}
   431→          sub={`${monthExpenses.length} transaction${
   432→            monthExpenses.length === 1 ? "" : "s"
   433→          }`}
   434→          tone="primary"
   435→        />

   437→          icon={<IndianRupee className="h-4 w-4" />}
   438→          label="All-time"
   439→          value={formatINR(allTimeTotal)}

   441→            expenses.length === 1 ? "" : "s"
   442→          }`}
   443→          tone="teal"
